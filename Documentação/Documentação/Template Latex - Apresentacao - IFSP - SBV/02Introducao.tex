\section{Introdução}

\begin{frame}

    \frametitle{Introdução}

    \begin{itemize}
    	\item Os jogos tem um enorme importância econômica na industria do entretenimento e também cultural;
    	\item Ótimos para estudos acadêmicos;
    	\item Importante fazer escolhas de qual tecnologia;
    \end{itemize}
    
    Com base nessas informações, será feito um teste de hipóteses analisando e comparando diferentes tecnologias de desenvolvimento de software, utilizando o jogo Tetris.

\end{frame}


\begin{frame}

    \frametitle{Objetivos}

    \begin{itemize}
    
        \item Objetivo Geral
            \begin{itemize}
                \item Comparar quatro formas diferentes de implementação do jogo Tetris, por meio das
                tecnologias Godot, Java , C++ e Python.
            \end{itemize}
            
        \item Objetivos Específicos
            \begin{itemize}
                \item Desenvolver Tetris no motor Gráfico Godot;
                \item Desenvolver Tetris com a linguagem Java utilizando a biblioteca LibGDX;
                \item Desenvolver Tetris com a linguagem c++ utilizando a biblioteca Raylib;
                \item Desenvolver Tetris com a linguagem Python utilizando a biblioteca PyGame;
                \item Coletar dados de tempo de desenvolvimento e performance dos jogos de Tetris;
                \item  Comparar dados de tempo de desenvolvimento e performance dos jogos de Tetris;
            \end{itemize}
            
    \end{itemize}

\end{frame}
